# Flat Saree Two-Step Workflow Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the temporary standalone Saree Try-On feature with a "Flat Saree" garment type in the standard studio wizard, where generation runs as two ComfyUI passes — a one-time internal mannequin-draping step (free, hidden) followed by one per-pose finalization job per selected pose (standard credit each, reusing the mannequin image as the garment input).

**Architecture:** Three new columns on `garment_subcategories` (`requiresMannequinStep`, `mannequinWorkflowTemplateId`, `sareeStep2WorkflowTemplateId`) gate the whole feature. A new job creator (`createSareeMannequinJob`) creates the hidden 0-credit step-1 job; the existing `createJob()` gains a `mannequinJobId` input alternative to `upperGarmentKey` for step-2 job creation, resolving it to the step-1 job's output image via a new ownership helper. The dispatcher gets a renamed/adapted mannequin handler (replacing `processSareeJob`) and a small override in the existing per-pose workflow-resolution code so every look in a flat-saree job uses the garment type's fixed step-2 workflow instead of the normal per-pose lookup. No new dispatcher coordination — step 1 and every step-2 job are independent queue-processed rows; sequencing lives in the frontend (create step-1, wait for `COMPLETED` via the existing SSE stream, then create the batch).

**Tech Stack:** Fastify 5 + Zod (`@tryme/types`), Drizzle ORM/Postgres, Redis Streams, Next.js 15 (studio wizard), Vite/React admin panel, Vitest (integration tests against the shared docker-compose Postgres/Redis/MinIO — no testcontainers).

## Global Constraints

- pnpm workspaces only — never introduce npm/yarn lockfiles.
- ESM only (`"type": "module"`), Node 20+, TypeScript 5.6.
- No `console.log` — use the existing `jobLog`/`app.log` pino loggers.
- Integration tests reuse the shared docker-compose Postgres/Redis/MinIO (`pnpm docker:up` must be running) — no testcontainers. `apps/api/vitest.config.ts` excludes `test/integration/**` from the default `pnpm --filter @tryme/api test` run; run integration tests with an override config (Task 6's Step 2 shows the exact command).
- `apps/dispatcher` integration tests use `apps/dispatcher/test/helpers/containers.ts`'s `setupTestEnv()` + `apps/dispatcher/test/helpers/comfy-mock.ts`'s `startComfyMock()` — same no-testcontainers rule.
- Schema changes go through `pnpm db:generate` (drizzle-kit, auto-generates migration SQL + journal + snapshot from the schema diff) then `pnpm db:migrate` — never hand-author migration SQL/journal files.
- Credit deduct + job insert is one Postgres transaction (existing invariant) — the new step-1 mannequin job creator charges 0 credits and skips `atomicDeduct` entirely (no ledger entry needed for a zero-amount charge).
- `@tryme/db` exports `* as schema` from `packages/db/src/index.ts` — never add a duplicate `schema` re-export.
- Only commit when a task's own step says to — this plan's commits are scoped per task already.

```ts
import type { DB } from '@tryme/db';
import { schema } from '@tryme/db';
import { jobsCreatedTotal } from '@tryme/observability';
import type { CreateSareeMannequinJobRequest } from '@tryme/types';
import { and, eq } from 'drizzle-orm';
import type { FastifyInstance } from 'fastify';
import type { z } from 'zod';
import { AppError } from '../../lib/errors.js';
import { assertOwnsUploadKey } from './create.js';

export async function createSareeMannequinJob(
  app: FastifyInstance,
  userId: string,
  body: z.infer<typeof CreateSareeMannequinJobRequest>,
) {
  const { garmentTypeId, garmentKey, faceId } = body;

  await assertOwnsUploadKey(app, userId, garmentKey);

  const [garmentType] = await app.db
    .select({
      isActive: schema.garmentSubcategories.isActive,
      requiresMannequinStep: schema.garmentSubcategories.requiresMannequinStep,
      mannequinWorkflowTemplateId: schema.garmentSubcategories.mannequinWorkflowTemplateId,
    })
    .from(schema.garmentSubcategories)
    .where(eq(schema.garmentSubcategories.id, garmentTypeId));
  if (!garmentType?.isActive || !garmentType.requiresMannequinStep) {
    throw new AppError('BAD_CATALOG', 400, 'garment type does not use a mannequin step');
  }
  if (!garmentType.mannequinWorkflowTemplateId) {
    throw new AppError('CONFIG', 400, 'garment type missing step-1 workflow configuration');
  }

  const [face] = await app.db
    .select({ id: schema.modelFaces.id })
    .from(schema.modelFaces)
    .where(and(eq(schema.modelFaces.id, faceId), eq(schema.modelFaces.isActive, true)));
  if (!face) throw new AppError('BAD_CATALOG', 400, 'face not found or inactive');

  const [[user], [planRow]] = await Promise.all([
    app.db.select().from(schema.users).where(eq(schema.users.id, userId)),
    app.db
      .select({
        queueStream: schema.creditPlans.queueStream,
        watermark: schema.creditPlans.watermark,
      })
      .from(schema.users)
      .innerJoin(schema.creditPlans, eq(schema.users.tier, schema.creditPlans.slug))
      .where(eq(schema.users.id, userId)),
  ]);
  if (!user || user.isBanned) throw new AppError('FORBIDDEN', 403, 'banned');

  const queueStream: string = planRow?.queueStream ?? 'normal';
  const priority = queueStream === 'priority';
  const watermark: boolean = planRow?.watermark ?? false;

  const job = await app.db.transaction(async (tx) => {
    const [newJob] = await tx
      .insert(schema.jobs)
      .values({
        userId,
        status: 'QUEUED',
        priority,
        queueStream,
        watermark,
        creditsCharged: 0,
        source: 'saree_mannequin',
      })
      .returning();
    await tx.insert(schema.jobInputs).values({
      jobId: newJob.id,
      upperGarmentKey: garmentKey,
      faceId,
      garmentTypeId,
      params: { kind: 'saree_mannequin' },
    });
    return newJob;
  });

  const stream = `jobs:${queueStream}`;
  try {
    await app.redis.xadd(stream, 'MAXLEN', '~', 10000, '*', 'jobId', job.id, 'userId', userId);
    jobsCreatedTotal.inc({ priority: queueStream, kind: 'saree_mannequin' });
  } catch (err) {
    app.log.error({ err, jobId: job.id }, 'redis xadd failed — mannequin job marked failed');
    await app.db
      .update(schema.jobs)
      .set({ status: 'FAILED', errorCode: 'ENQUEUE_FAIL' })
      .where(eq(schema.jobs.id, job.id));
    throw new AppError('ENQUEUE_FAIL', 503, 'queue unavailable');
  }

  return { jobId: job.id };
}
```

Note: `jobs.catalogueId` is nullable (every other job creator sets it via `randomUUID()`, but a hidden internal job that's excluded from every catalogue listing has no use for one) — leave it unset.

```ts
/**
 * Resolves a completed saree-mannequin job (see createSareeMannequinJob) to its
 * output R2 key, for use as a step-2 job's upperGarmentKey. Mirrors
 * assertOwnsUploadKey's ownership check, but against a job's own output rather

export async function resolveMannequinGarmentKey(
  app: FastifyInstance,
  userId: string,
  mannequinJobId: string,
): Promise<string> {
  const [row] = await app.db
    .select({
      userId: schema.jobs.userId,
      status: schema.jobs.status,
      kind: sql<string>`${schema.jobInputs.params}->>'kind'`.as('kind'),
    })
    .from(schema.jobs)
    .innerJoin(schema.jobInputs, eq(schema.jobInputs.jobId, schema.jobs.id))
    .where(eq(schema.jobs.id, mannequinJobId));
  if (!row || row.userId !== userId) {
    throw new AppError('FORBIDDEN', 403, 'mannequin job not owned by caller');
  }
  if (row.kind !== 'saree_mannequin') {
    throw new AppError('VALIDATION', 400, 'job is not a mannequin generation job');
  }
  if (row.status !== 'COMPLETED') {
    throw new AppError('VALIDATION', 400, 'mannequin generation not yet complete');
  }
  return keys.output(mannequinJobId);
}
```
